from dataclasses import dataclass


@dataclass(frozen=True)
class AssuranceConfig:
    """
    Configuration for assurance signal fusion.

    The weights must sum to 1.0.
    """

    behavior_weight: float = 0.60
    face_weight: float = 0.25
    context_weight: float = 0.15

    def __post_init__(self) -> None:
        total = (
            self.behavior_weight
            + self.face_weight
            + self.context_weight
        )

        if abs(total - 1.0) > 1e-9:
            raise ValueError(
                "Assurance signal weights must sum to 1.0."
            )

        for name, value in (
            ("behavior_weight", self.behavior_weight),
            ("face_weight", self.face_weight),
            ("context_weight", self.context_weight),
        ):
            if not 0.0 <= value <= 1.0:
                raise ValueError(
                    f"{name} must be between 0.0 and 1.0."
                )


@dataclass(frozen=True)
class BehavioralConfig:
    """
    Configuration for the Behavioral Signals Engine.

    The behavioral engine deliberately uses conservative
    bounds and robust statistical controls.
    """

    required_observations: int = 50

    minimum_feature_observations: int = 5

    minimum_observation_quality: float = 0.20

    maximum_features_missing_ratio: float = 0.60

    maximum_update_z_score: float = 4.0

    maximum_temporal_gap_seconds: float = 300.0

    temporal_decay_seconds: float = 60.0

    consistency_threshold: float = 0.75

    uncertainty_threshold: float = 0.45

    def __post_init__(self) -> None:
        if self.required_observations < 1:
            raise ValueError(
                "required_observations must be at least 1."
            )

        if self.minimum_feature_observations < 1:
            raise ValueError(
                "minimum_feature_observations must be at least 1."
            )

        if not 0.0 <= self.minimum_observation_quality <= 1.0:
            raise ValueError(
                "minimum_observation_quality must be "
                "between 0.0 and 1.0."
            )

        if not 0.0 <= self.maximum_features_missing_ratio <= 1.0:
            raise ValueError(
                "maximum_features_missing_ratio must be "
                "between 0.0 and 1.0."
            )

        if self.maximum_update_z_score <= 0.0:
            raise ValueError(
                "maximum_update_z_score must be greater than 0."
            )

        if self.maximum_temporal_gap_seconds <= 0.0:
            raise ValueError(
                "maximum_temporal_gap_seconds must be "
                "greater than 0."
            )

        if self.temporal_decay_seconds <= 0.0:
            raise ValueError(
                "temporal_decay_seconds must be greater than 0."
            )

        if not 0.0 <= self.uncertainty_threshold <= 1.0:
            raise ValueError(
                "uncertainty_threshold must be "
                "between 0.0 and 1.0."
            )

        if not 0.0 <= self.consistency_threshold <= 1.0:
            raise ValueError(
                "consistency_threshold must be "
                "between 0.0 and 1.0."
            )

        if (
            self.uncertainty_threshold
            >= self.consistency_threshold
        ):
            raise ValueError(
                "uncertainty_threshold must be lower "
                "than consistency_threshold."
            )


@dataclass(frozen=True)
class PolicyConfig:
    """
    Centralized policy thresholds.

    The policy has three possible outcomes:

        assurance >= allow_threshold
            -> allow

        assurance >= reverify_threshold
            -> reverify

        assurance < reverify_threshold
            -> lock
    """

    allow_threshold: float = 0.75
    reverify_threshold: float = 0.45

    max_reverify_attempts: int = 3

    def __post_init__(self) -> None:
        if not 0.0 <= self.reverify_threshold <= 1.0:
            raise ValueError(
                "reverify_threshold must be between 0.0 and 1.0."
            )

        if not 0.0 <= self.allow_threshold <= 1.0:
            raise ValueError(
                "allow_threshold must be between 0.0 and 1.0."
            )

        if self.reverify_threshold >= self.allow_threshold:
            raise ValueError(
                "reverify_threshold must be lower "
                "than allow_threshold."
            )

        if self.max_reverify_attempts < 1:
            raise ValueError(
                "max_reverify_attempts must be at least 1."
            )

def get_session(
    session_id: str | None,
):
    """
    Resolve an authenticated session.

    This helper only resolves the session object.
    Lifecycle validation is handled separately by
    ensure_valid_session().
    """

    if session_id is None:
        raise HTTPException(
            status_code=401,
            detail="Authentication required.",
        )

    session = session_store.get(
        session_id
    )

    if session is None:
        raise HTTPException(
            status_code=401,
            detail="Invalid session.",
        )

    return session


def ensure_valid_session(
    session_id: str | None,
):
    """
    Resolve and validate an authenticated session.

    SessionStore owns inactivity expiration and lifecycle
    transitions.

    Locked sessions are rejected here because this helper is
    used by protected operations that require an active
    session.
    """

    session = get_session(
        session_id
    )

    session_store.ensure_valid(
        session
    )

    if session.is_locked:
        raise HTTPException(
            status_code=401,
            detail=(
                "Session is locked. "
                + (
                    session.lock_reason
                    or
                    "Authentication is required."
                )
            ),
        )

    return session

@app.post("/api/login")
def login(
    request: LoginRequest,
):
    if (
        request.username
        != DEMO_USERNAME
        or request.password
        != DEMO_PASSWORD
    ):
        audit(
            AuditEventType.LOGIN_FAILURE,
            metadata={
                "username": request.username,
            },
        )

        raise HTTPException(
            status_code=401,
            detail="Invalid username or password.",
        )

    if request.username not in profiles:
        profiles[
            request.username
        ] = UserProfile(
            user_id=request.username
        )

    profile = profiles[
        request.username
    ]

    session_id = (
        secrets.token_urlsafe(32)
    )

    session_store.create(
        session_id=session_id,
        user_id=request.username,
    )

    audit(
        AuditEventType.LOGIN_SUCCESS,
        user_id=request.username,
        session_id=session_id,
    )

    cookie = (
        f"{config.session.cookie_name}="
        f"{session_id}; "
        f"Path={config.session.session_cookie_path}; "
    )

    if config.session.session_cookie_http_only:
        cookie += "HttpOnly; "

    cookie += (
        f"SameSite="
        f"{config.session.session_cookie_same_site}"
    )

    if config.session.session_cookie_secure:
        cookie += "; Secure"

    return JSONResponse(
        content={
            "authenticated": True,
            "user_id": request.username,
            "profile_state": profile.state,
            "session_status": "active",
        },
        headers={
            "Set-Cookie": cookie,
        },
    )

@app.post("/api/logout")
def logout(
    session_id: str | None = Cookie(
        default=None,
        alias=config.session.cookie_name,
    ),
):
    """
    Explicitly terminate the current authenticated session.

    Logout is terminal for the current session. The server-side
    session is locked and the client session cookie is cleared.
    """

    session = get_session(
        session_id
    )

    session_store.ensure_valid(
        session
    )

    if session.is_locked:
        raise HTTPException(
            status_code=401,
            detail=(
                "Session is locked. "
                + (
                    session.lock_reason
                    or
                    "Authentication is required."
                )
            ),
        )

    session_store.logout(
        session
    )

    response = JSONResponse(
        content={
            "authenticated": False,
            "logged_out": True,
            "session_status": "logged_out",
            "message": (
                "Session logged out successfully."
            ),
        }
    )

    response.delete_cookie(
        key=config.session.cookie_name,
        path=config.session.session_cookie_path,
    )

    return response

@app.post("/api/session/reverify")
def reverify_session(
    request: ReverificationRequest,
    session_id: str | None = Cookie(
        default=None,
        alias=config.session.cookie_name,
    ),
):
    # ======================================================
    # 1. SESSION + EXPIRATION
    # ======================================================

    session = ensure_valid_session(
        session_id
    )

    # ======================================================
    # 2. NOT CURRENTLY REVERIFYING
    # ======================================================

    if not session.requires_reverification:
        return {
            "verified": True,
            "session_state": (
                session.state.value
            ),
            "message": (
                "Session does not currently "
                "require reverification."
            ),
        }

    # ======================================================
    # 3. FAILED VERIFICATION
    # ======================================================

    if not request.verified:
        should_lock = (
            policy_engine
            .should_lock_after_failed_reverification(
                session.reverify_count
            )
        )

        if should_lock:
            reason = (
                "Maximum reverification attempts "
                "exceeded."
            )

            session_store.lock(
                session,
                reason=reason,
            )

            return {
                "verified": False,
                "session_state": (
                    session.state.value
                ),
                "session_active": False,
                "reverify_count": (
                    session.reverify_count
                ),
                "action": "lock",
                "message": (
                    "Reverification failed too many "
                    "times. Session has been locked."
                ),
            }

        session_store.mark_reverify(
            session,
            reason=(
                "Identity verification failed."
            ),
        )

        audit(
            AuditEventType.REVERIFICATION_FAILURE,
            user_id=session.user_id,
            session_id=session.session_id,
            action="reverify",
            reason=(
                "Identity verification failed."
            ),
        )

        return {
            "verified": False,
            "session_state": (
                session.state.value
            ),
            "session_active": (
                session.active
            ),
            "reverify_count": (
                session.reverify_count
            ),
            "action": "reverify",
            "message": (
                "Identity verification failed. "
                "Reverification is still required."
            ),
        }

    # ======================================================
    # 4. SUCCESSFUL VERIFICATION
    # ======================================================

    session_store.activate(
        session,
        reason=(
            "Identity verification successful."
        ),
    )

    audit(
        AuditEventType.REVERIFICATION_SUCCESS,
        user_id=session.user_id,
        session_id=session.session_id,
        action="allow",
        reason=(
            "Identity verification successful."
        ),
    )

    return {
        "verified": True,
        "session_state": (
            session.state.value
        ),
        "session_active": (
            session.active
        ),
        "reverify_count": (
            session.reverify_count
        ),
        "message": (
            "Identity verification "
            "successful. Session restored."
        ),
    }

@app.post("/api/behavior/observation")
def behavior_observation(
    observation: BehaviorObservation,
    session_id: str | None = Cookie(
        default=None,
        alias=config.session.cookie_name,
    ),
):
    # ======================================================
    # 1. SESSION + EXPIRATION
    # ======================================================

    session = ensure_valid_session(
        session_id
    )

    # ======================================================
    # 2. REVERIFICATION
    # ======================================================

    if session.requires_reverification:
        raise HTTPException(
            status_code=403,
            detail=(
                "Identity verification required "
                "before submitting behavior."
            ),
        )

    # ======================================================
    # 3. ACTIVITY
    # ======================================================

    session_store.touch(
        session
    )

    # ======================================================
    # 4. PROFILE
    # ======================================================

    user_id = session.user_id

    if user_id not in profiles:
        profiles[user_id] = UserProfile(
            user_id=user_id
        )

    profile = profiles[
        user_id
    ]

    # ======================================================
    # 5. AUDIT OBSERVATION
    # ======================================================

    audit(
        AuditEventType.BEHAVIOR_OBSERVATION,
        user_id=user_id,
        session_id=session.session_id,
        metadata={
            "profile_state": str(
                profile.state
            ),
        },
    )

    # ======================================================
    # 6. LEARNING
    # ======================================================

    if profile.state != "established":
        profile = (
            learning_engine.update_profile(
                profile,
                observation,
            )
        )

        profiles[user_id] = profile

        progress = (
            profile.observation_count
            /
            profile.required_observations
        )

        progress = min(
            progress,
            1.0,
        )

        return {
            "mode": "learning",
            "profile_state": (
                profile.state
            ),
            "observation_count": (
                profile.observation_count
            ),
            "required_observations": (
                profile.required_observations
            ),
            "progress": progress,
            "behavior_score": None,
            "assurance_score": None,
            "risk_score": None,
            "action": "learning",
            "session_state": (
                session.state.value
            ),
            "session_active": (
                session.active
            ),
            "reason": (
                "Behavioral profile is still "
                "being established."
            ),
        }

    # ======================================================
    # 7. ASSURANCE + POLICY
    # ======================================================

    decision = (
        assurance_engine.evaluate(
            profile,
            observation,
        )
    )

    # ======================================================
    # 8. STORE SCORES
    # ======================================================

    session.behavior_score = (
        decision.behavior_score
    )

    session.assurance_score = (
        decision.assurance_score
    )

    session.risk_score = (
        decision.risk_score
    )

    # ======================================================
    # 9. AUDIT ASSURANCE
    # ======================================================

    audit(
        AuditEventType.ASSURANCE_EVALUATED,
        user_id=user_id,
        session_id=session.session_id,
        action=decision.action.value,
        reason=decision.reason,
        assurance_score=(
            decision.assurance_score
        ),
        risk_score=(
            decision.risk_score
        ),
    )

    # ======================================================
    # 10. APPLY POLICY
    # ======================================================

    session_store.apply_action(
        session,
        decision.action,
        decision.reason,
    )

    # ======================================================
    # 11. RESPONSE
    # ======================================================

    return {
        "mode": "verification",
        "profile_state": (
            profile.state
        ),
        "observation_count": (
            profile.observation_count
        ),
        "behavior_score": (
            decision.behavior_score
        ),
        "face_score": (
            decision.face_score
        ),
        "context_score": (
            decision.context_score
        ),
        "assurance_score": (
            decision.assurance_score
        ),
        "risk_score": (
            decision.risk_score
        ),
        "action": (
            decision.action.value
        ),
        "session_state": (
            session.state.value
        ),
        "session_active": (
            session.active
        ),
        "reverify_count": (
            session.reverify_count
        ),
        "reason": (
            decision.reason
        ),
    }

from threading import Lock

from app.config import AuditConfig

from app.models.audit import (
    AuditEvent,
)


class AuditLogger:
    """
    Small in-memory audit logger.

    The interface deliberately does not depend on storage.

    A future implementation can persist events to:
    - SQLite
    - PostgreSQL
    - an external SIEM
    - an append-only event store
    """

    def __init__(
        self,
        config: AuditConfig | None = None,
    ) -> None:

        self.config = config or AuditConfig()

        self._events: list[AuditEvent] = []

        self._lock = Lock()

    def record(
        self,
        event: AuditEvent,
    ) -> None:

        if not self.config.enabled:
            return

        with self._lock:

            self._events.append(
                event
            )

            overflow = (
                len(self._events)
                - self.config.max_events
            )

            if overflow > 0:

                del self._events[
                    :overflow
                ]

    def list_events(self) -> list[AuditEvent]:

        with self._lock:

            return list(
                self._events
            )

    def clear(self) -> None:

        with self._lock:

            self._events.clear()

    def count(self) -> int:

        with self._lock:

            return len(
                self._events
            )

import math
from typing import Mapping

from app.models.signals import (
    BehaviorObservation,

class BehavioralSignalsEngine:
    """
    Converts privacy-conscious behavioral measurements into
    normalized behavioral assurance signals.

    This layer deliberately does NOT process or store:

    - raw keystrokes
    - passwords
    - page contents
    - mouse coordinates
    - screenshots

    It operates only on numerical behavioral measurements.

    Supported measurements:

    - typing_speed
    - typing_interval_variance
    - mouse_speed
    - click_interval
    - navigation_interval
    """

    FEATURE_NAMES = (
        "typing_speed",
        "typing_interval_variance",
        "mouse_speed",
        "click_interval",
        "navigation_interval",
    )

    def __init__(
        self,
        minimum_features: int = 1,
    ) -> None:

        if minimum_features < 1:
            raise ValueError(
                "minimum_features must be at least 1."
            )

        if minimum_features > len(
            self.FEATURE_NAMES
        ):
            raise ValueError(
                "minimum_features cannot exceed "
                "the number of supported behavioral features."
            )

        self.minimum_features = (
            minimum_features
        )

    # ==========================================================
    # OBSERVATION CONSTRUCTION
    # ==========================================================

    def build_observation(
        self,
        *,
        typing_speed: float | None = None,
        typing_interval_variance: float | None = None,
        mouse_speed: float | None = None,
        click_interval: float | None = None,
        navigation_interval: float | None = None,
    ) -> BehaviorObservation:
        """
        Construct a sanitized BehaviorObservation.

        Invalid numerical values such as NaN, infinity, and
        negative measurements are discarded rather than passed
        into the assurance pipeline.
        """

        values = {
            "typing_speed": typing_speed,
            "typing_interval_variance": (
                typing_interval_variance
            ),
            "mouse_speed": mouse_speed,
            "click_interval": click_interval,
            "navigation_interval": (
                navigation_interval
            ),
        }

        sanitized: dict[str, float] = {}

        for name, value in values.items():

            cleaned = self._sanitize_value(
                value
            )

            if cleaned is not None:
                sanitized[name] = cleaned

        return BehaviorObservation(
            **sanitized
        )

    # ==========================================================
    # MAPPING SUPPORT
    # ==========================================================

    def from_mapping(
        self,
        values: Mapping[str, float | None],
    ) -> BehaviorObservation:
        """
        Build an observation from a mapping.

        Unknown fields are ignored.

        This provides a clean boundary between an external
        telemetry collector and the assurance engine.
        """

        supported = {
            name: values.get(name)
            for name in self.FEATURE_NAMES
        }

        return self.build_observation(
            **supported
        )

    # ==========================================================
    # QUALITY
    # ==========================================================

    def calculate_quality(
        self,
        observation: BehaviorObservation,
    ) -> float:
        """
        Calculate observation quality based on feature
        availability.

        The result is always between 0.0 and 1.0.
        """

        return max(
            0.0,
            min(
                1.0,
                observation.completeness,
            ),
        )

    # ==========================================================
    # SIGNAL GENERATION
    # ==========================================================

    def generate(
        self,
        observation: BehaviorObservation,
    ) -> SignalResult:
        """
        Generate a behavioral signal from an observation.

        This signal represents the quality and availability
        of behavioral evidence.

        Actual behavioral consistency is evaluated later by
        BehavioralVerifier against the learned profile.
        """

        quality = self.calculate_quality(
            observation
        )

        feature_count = (
            observation.feature_count
        )

        if feature_count < self.minimum_features:

            return SignalResult(
                source="behavioral_signals",
                score=0.5,
                confidence=0.0,
                quality=quality,
                status=SignalStatus.UNAVAILABLE,
            )

        if quality < 0.5:

            status = SignalStatus.UNCERTAIN

            confidence = 0.3

        else:

            status = SignalStatus.CONSISTENT

            confidence = min(
                1.0,
                0.5 + (quality * 0.5),
            )

        return SignalResult(
            source="behavioral_signals",
            score=1.0,
            confidence=confidence,
            quality=quality,
            status=status,
        )

    # ==========================================================
    # FEATURE EXTRACTION
    # ==========================================================

    def extract_features(
        self,
        observation: BehaviorObservation,
    ) -> dict[str, float]:
        """
        Return only available numerical behavioral features.

        No raw user content is returned.
        """

        values = observation.model_dump(
            exclude_none=True
        )

        result: dict[str, float] = {}

        for name, value in values.items():

            cleaned = self._sanitize_value(
                value
            )

            if cleaned is not None:
                result[name] = cleaned

        return result

    # ==========================================================
    # VALIDATION
    # ==========================================================

    @staticmethod
    def _sanitize_value(
        value: float | None,
    ) -> float | None:
        """
        Validate one behavioral measurement.

        Returns None for unusable values.
        """

        if value is None:
            return None

        try:
            numeric_value = float(value)

        except (
            TypeError,
            ValueError,
        ):
            return None

        if not math.isfinite(
            numeric_value
        ):
            return None

        if numeric_value < 0.0:
            return None

        return numeric_value

import math

class BehavioralLearningEngine:
    """
    Incrementally learns a behavioral profile.

    Important security properties:

    - Raw history is not retained.
    - Statistics are updated online.
    - Extreme observations are prevented from dominating
      established profiles.
    - Missing features do not reset existing statistics.
    - Learning stops treating the profile as untrusted once
      sufficient observations have been collected.
    """

    def __init__(
        self,
        required_observations: int = 50,
        behavioral_config: BehavioralConfig | None = None,
    ) -> None:

        self.config = (
            behavioral_config
            or BehavioralConfig(
                required_observations=(
                    required_observations
                )
            )
        )

        self.required_observations = (
            self.config.required_observations
        )

    def update_profile(
        self,
        profile: UserProfile,
        observation: BehaviorObservation,
    ) -> UserProfile:

        profile.required_observations = (
            self.required_observations
        )

        values = observation.feature_values()

        if not values:
            return profile

        accepted_features = 0

        for feature_name, value in values.items():

            if not math.isfinite(value):
                continue

            statistics = profile.features.get(
                feature_name
            )

            if statistics is None:
                statistics = FeatureStatistics()

                profile.features[
                    feature_name
                ] = statistics

            accepted = self._update_feature(
                statistics,
                value,
            )

            if accepted:
                accepted_features += 1

        if accepted_features == 0:
            return profile

        profile.observation_count += 1

        profile.last_updated = (
            datetime.now(timezone.utc)
        )

        profile.last_behavior_observation_at = (
            observation.timestamp
        )

        if (
            profile.observation_count
            >= self.required_observations
        ):
            profile.state = "established"

            if profile.established_at is None:
                profile.established_at = (
                    datetime.now(timezone.utc)
                )

        else:
            profile.state = "learning"

        return profile

    def _update_feature(
        self,
        statistics: FeatureStatistics,
        value: float,
    ) -> bool:

        old_count = statistics.count

        if old_count >= (
            self.config.minimum_feature_observations
        ):

            standard_deviation = math.sqrt(
                max(
                    statistics.variance,
                    0.0,
                )
            )

            standard_deviation = max(
                standard_deviation,
                0.001,
            )

            z_score = abs(
                value - statistics.mean
            ) / standard_deviation

            if (
                z_score
                > self.config.maximum_update_z_score
            ):
                return False

        new_count = old_count + 1

        if old_count == 0:

            statistics.count = 1
            statistics.mean = value
            statistics.variance = 0.0
            statistics.minimum = value
            statistics.maximum = value
            statistics.last_updated = (
                datetime.now(timezone.utc)
            )

            return True

        delta = (
            value
            - statistics.mean
        )

        new_mean = (
            statistics.mean
            + delta / new_count
        )

        delta2 = (
            value
            - new_mean
        )

        old_m2 = (
            statistics.variance
            * old_count
        )

        new_m2 = (
            old_m2
            + delta * delta2
        )

        new_variance = (
            new_m2 / new_count
        )

        statistics.count = new_count

        statistics.mean = new_mean

        statistics.variance = max(
            0.0,
            new_variance,
        )

        if statistics.minimum is None:
            statistics.minimum = value
        else:
            statistics.minimum = min(
                statistics.minimum,
                value,
            )

        if statistics.maximum is None:
            statistics.maximum = value
        else:
            statistics.maximum = max(
                statistics.maximum,
                value,
            )

        statistics.last_updated = (
            datetime.now(timezone.utc)
        )

        return True

import math
from datetime import datetime, timezone

class BehavioralSignalsEngine:
    """
    Behavioral Signals Engine.

    Converts behavioral observations and a learned profile
    into a bounded SignalResult.

    Security goals:

    - complete feature mediation
    - no trust in raw behavioral input
    - bounded numerical output
    - missing-feature awareness
    - temporal consistency
    - outlier resistance
    - conservative confidence handling
    """

    def __init__(
        self,
        config: BehavioralConfig | None = None,
    ) -> None:

        self.config = (
            config
            or BehavioralConfig()
        )

    def evaluate(
        self,
        profile: UserProfile,
        observation: BehaviorObservation,
    ) -> SignalResult:

        values = observation.feature_values()

        if not values:

            return SignalResult(
                source="behavior",
                score=0.5,
                confidence=0.0,
                quality=0.0,
                status=SignalStatus.UNAVAILABLE,
            )

        finite_values = {
            name: value
            for name, value in values.items()
            if math.isfinite(value)
        }

        if not finite_values:

            return SignalResult(
                source="behavior",
                score=0.5,
                confidence=0.0,
                quality=0.0,
                status=SignalStatus.UNAVAILABLE,
            )

        feature_scores: list[float] = []

        feature_weights: list[float] = []

        for (
            feature_name,
            value,
        ) in finite_values.items():

            statistics = profile.features.get(
                feature_name
            )

            if statistics is None:
                continue

            if (
                statistics.count
                < self.config.minimum_feature_observations
            ):
                continue

            score = self._feature_score(
                statistics.mean,
                statistics.variance,
                value,
            )

            reliability = self._feature_reliability(
                statistics.count
            )

            feature_scores.append(
                score
            )

            feature_weights.append(
                reliability
            )

        if not feature_scores:

            return SignalResult(
                source="behavior",
                score=0.5,
                confidence=0.25,
                quality=self._observation_quality(
                    profile,
                    finite_values,
                ),
                status=SignalStatus.UNCERTAIN,
            )

        behavioral_score = self._weighted_average(
            feature_scores,
            feature_weights,
        )

        observation_quality = (
            self._observation_quality(
                profile,
                finite_values,
            )
        )

        temporal_score = (
            self._temporal_consistency(
                profile,
                observation.timestamp,
            )
        )

        combined_score = (
            behavioral_score
            * (0.80 + 0.20 * temporal_score)
        )

        combined_score = max(
            0.0,
            min(
                1.0,
                combined_score,
            ),
        )

        confidence = self._confidence(
            profile,
            len(feature_scores),
            observation_quality,
            temporal_score,
        )

        status = self._status(
            combined_score,
            confidence,
        )

        return SignalResult(
            source="behavior",
            score=combined_score,
            confidence=confidence,
            quality=observation_quality,
            status=status,
        )

    def _feature_score(
        self,
        mean: float,
        variance: float,
        value: float,
    ) -> float:

        standard_deviation = math.sqrt(
            max(
                variance,
                0.0,
            )
        )

        deviation = abs(
            value - mean
        )

        if deviation == 0.0:
            return 1.0

        standard_deviation = max(
            standard_deviation,
            0.001,
        )

        z_score = (
            deviation
            / standard_deviation
        )

        # Huber-like transition:
        #
        # small deviations remain smooth,
        # large deviations become increasingly
        # suspicious without numerical instability.
        if z_score <= 2.0:

            score = math.exp(
                -0.5
                * (
                    z_score ** 2
                )
            )

        else:

            score = math.exp(
                -2.0
                - (
                    0.5
                    * (
                        z_score
                        - 2.0
                    )
                )
            )

        return max(
            0.0,
            min(
                1.0,
                score,
            ),
        )

    @staticmethod
    def _feature_reliability(
        count: int,
    ) -> float:

        if count <= 0:
            return 0.0

        # Reliability approaches 1 asymptotically.
        return min(
            1.0,
            math.log1p(count)
            / math.log1p(50),
        )

    def _observation_quality(
        self,
        profile: UserProfile,
        values: dict[str, float],
    ) -> float:

        known_features = 0

        for feature_name in values:

            if feature_name in profile.features:
                known_features += 1

        total_expected = max(
            len(profile.features),
            1,
        )

        known_ratio = (
            known_features
            / total_expected
        )

        finite_ratio = (
            len(values)
            / max(
                len(
                    profile.features
                ),
                len(values),
                1,
            )
        )

        quality = (
            0.60 * known_ratio
            + 0.40 * finite_ratio
        )

        return max(
            0.0,
            min(
                1.0,
                quality,
            ),
        )

    def _temporal_consistency(
        self,
        profile: UserProfile,
        timestamp: datetime,
    ) -> float:

        previous = (
            profile.last_behavior_observation_at
        )

        if previous is None:
            return 1.0

        current = timestamp

        if current.tzinfo is None:
            current = current.replace(
                tzinfo=timezone.utc
            )

        if previous.tzinfo is None:
            previous = previous.replace(
                tzinfo=timezone.utc
            )

        gap = abs(
            (
                current
                - previous
            ).total_seconds()
        )

        if gap <= 0.0:
            return 0.5

        if (
            gap
            > self.config.maximum_temporal_gap_seconds
        ):
            return 0.75

        return math.exp(
            -gap
            / self.config.temporal_decay_seconds
        )

    @staticmethod
    def _weighted_average(
        scores: list[float],
        weights: list[float],
    ) -> float:

        if not scores:
            return 0.5

        total_weight = sum(weights)

        if total_weight <= 0.0:
            return sum(scores) / len(scores)

        return (
            sum(
                score * weight
                for score, weight
                in zip(
                    scores,
                    weights,
                )
            )
            / total_weight
        )

    @staticmethod
    def _confidence(
        profile: UserProfile,
        feature_count: int,
        quality: float,
        temporal_score: float,
    ) -> float:

        maturity = min(
            1.0,
            profile.observation_count
            / max(
                profile.required_observations,
                1,
            ),
        )

        coverage = min(
            1.0,
            feature_count / 3.0,
        )

        confidence = (
            0.40 * maturity
            + 0.25 * coverage
            + 0.20 * quality
            + 0.15 * temporal_score
        )

        return max(
            0.0,
            min(
                1.0,
                confidence,
            ),
        )

    def _status(
        self,
        score: float,
        confidence: float,
    ) -> SignalStatus:

        if confidence < 0.30:
            return SignalStatus.UNCERTAIN

        if score >= self.config.consistency_threshold:
            return SignalStatus.CONSISTENT

        if score >= self.config.uncertainty_threshold:
            return SignalStatus.UNCERTAIN

        return SignalStatus.MISMATCH

from __future__ import annotations

from app.models.assurance import AssuranceResult
from app.models.signals import (
    SignalResult,
    SignalStatus,
)


class RiskFusionEngine:
    """
    Combines available assurance signals into a normalized
    assurance and risk result.

    Important properties:

    - Signal score is affected by confidence and quality.
    - UNAVAILABLE signals do not contribute negative evidence.
    - Available signals retain their configured weights.
    - Remaining weights are renormalized when a signal is
      unavailable.
    """

    def __init__(
        self,
        behavior_weight: float = 0.60,
        face_weight: float = 0.25,
        context_weight: float = 0.15,
    ) -> None:

        weights = (
            (
                "behavior_weight",
                behavior_weight,
            ),
            (
                "face_weight",
                face_weight,
            ),
            (
                "context_weight",
                context_weight,
            ),
        )

        for name, value in weights:

            if not 0.0 <= value <= 1.0:
                raise ValueError(
                    f"{name} must be between 0.0 and 1.0."
                )

        if sum(
            value
            for _, value in weights
        ) <= 0.0:

            raise ValueError(
                "At least one signal weight must "
                "be greater than 0."
            )

        self.behavior_weight = behavior_weight
        self.face_weight = face_weight
        self.context_weight = context_weight

    @staticmethod
    def _is_available(
        signal: SignalResult | None,
    ) -> bool:

        return (
            signal is not None
            and signal.status
            != SignalStatus.UNAVAILABLE
        )

    def fuse(
        self,
        behavior: SignalResult | None = None,
        face: SignalResult | None = None,
        context: SignalResult | None = None,
    ) -> AssuranceResult:

        weighted_total = 0.0
        total_weight = 0.0

        signals = (
            (
                behavior,
                self.behavior_weight,
            ),
            (
                face,
                self.face_weight,
            ),
            (
                context,
                self.context_weight,
            ),
        )

        for signal, weight in signals:

            if not self._is_available(signal):
                continue

            if weight <= 0.0:
                continue

            weighted_total += (
                signal.effective_score
                * weight
            )

            total_weight += weight

        if total_weight <= 0.0:

            assurance_score = 0.0

        else:

            assurance_score = (
                weighted_total
                / total_weight
            )

        assurance_score = max(
            0.0,
            min(
                1.0,
                assurance_score,
            ),
        )

        risk_score = (
            1.0
            - assurance_score
        )

        return AssuranceResult(
            assurance_score=assurance_score,
            risk_score=risk_score,

            behavior_score=(
                behavior.score
                if behavior is not None
                else 0.0
            ),

            face_score=(
                face.score
                if face is not None
                else 0.0
            ),

            context_score=(
                context.score
                if context is not None
                else 0.0
            ),
        )

from collections import deque


class TemporalAnalyzer:

    def __init__(
        self,
        window_size: int = 5,
        anomaly_threshold: float = 0.30,
    ):
        self.window_size = window_size
        self.anomaly_threshold = anomaly_threshold

        self.history: dict[str, deque[float]] = {}
        self.anomaly_streaks: dict[str, int] = {}

    def update(
        self,
        session_id: str,
        risk_score: float,
    ) -> tuple[float, int]:

        if session_id not in self.history:
            self.history[session_id] = deque(
                maxlen=self.window_size
            )

        if session_id not in self.anomaly_streaks:
            self.anomaly_streaks[session_id] = 0

        history = self.history[session_id]

        history.append(risk_score)

        if risk_score >= self.anomaly_threshold:
            self.anomaly_streaks[session_id] += 1
        else:
            self.anomaly_streaks[session_id] = 0

        # More recent observations receive more weight.
        weights = range(1, len(history) + 1)

        weighted_sum = sum(
            risk * weight
            for risk, weight in zip(history, weights)
        )

        total_weight = sum(weights)

        temporal_risk = weighted_sum / total_weight

        return temporal_risk, self.anomaly_streaks[session_id]

    def clear(self, session_id: str):
        self.history.pop(session_id, None)
        self.anomaly_streaks.pop(session_id, None)

from datetime import datetime, timezone
from typing import Callable

from app.audit.logger import (
    AuditLogger,
)

from app.config import (
    SessionConfig,
)

from app.models.audit import (
    AuditEvent,
    AuditEventType,
)

from app.models.decision import (
    DecisionAction,
)

from app.models.session import (
    Session,
    SessionState,
)


class SessionStore:
    """
    In-memory session store.

    This class owns session lifecycle transitions.

    Lifecycle responsibilities include:
    - session creation
    - activity tracking
    - inactivity expiration
    - reverification transitions
    - activation
    - locking
    - logout
    - policy action application
    - lifecycle audit events

    The clock is injected so time-dependent behavior can be
    tested deterministically without patching datetime.now().
    """

    def __init__(
        self,
        audit_logger: AuditLogger | None = None,
        config: SessionConfig | None = None,
        clock: Callable[[], datetime] | None = None,
    ) -> None:
        self._sessions: dict[
            str,
            Session,
        ] = {}

        self._audit_logger = audit_logger

        self._config = (
            config
            or SessionConfig()
        )

        self._clock = (
            clock
            or self._default_clock
        )

    # ======================================================
    # INTERNAL HELPERS
    # ======================================================

    @staticmethod
    def _default_clock() -> datetime:
        """
        Return the current UTC time.
        """

        return datetime.now(
            timezone.utc
        )

    def _now(self) -> datetime:
        """
        Return the current time from the injected clock.
        """

        return self._clock()

    def _audit(
        self,
        event_type: AuditEventType,
        *,
        session: Session,
        action: str | None = None,
        reason: str | None = None,
    ) -> None:
        if self._audit_logger is None:
            return

        self._audit_logger.record(
            AuditEvent(
                event_type=event_type,
                user_id=session.user_id,
                session_id=session.session_id,
                action=action,
                reason=reason,
                assurance_score=(
                    session.assurance_score
                ),
                risk_score=(
                    session.risk_score
                ),
            )
        )

    # ======================================================
    # CREATE
    # ======================================================

    def create(
        self,
        session_id: str,
        user_id: str,
    ) -> Session:
        if session_id in self._sessions:
            raise ValueError(
                "Session already exists."
            )

        now = self._now()

        session = Session(
            session_id=session_id,
            user_id=user_id,
            created_at=now,
            last_activity_at=now,
            state_changed_at=now,
        )

        self._sessions[
            session_id
        ] = session

        self._audit(
            AuditEventType.SESSION_ACTIVATED,
            session=session,
            action="create",
            reason=(
                "Authenticated session created."
            ),
        )

        return session

    # ======================================================
    # GET
    # ======================================================

    def get(
        self,
        session_id: str,
    ) -> Session | None:
        return self._sessions.get(
            session_id
        )

    # ======================================================
    # TOUCH
    # ======================================================

    def touch(
        self,
        session: Session,
    ) -> Session:
        if session.is_locked:
            return session

        if self.is_expired(session):
            self.expire(session)
            return session

        session.last_activity_at = (
            self._now()
        )

        return session

    # ======================================================
    # EXPIRATION
    # ======================================================

    def is_expired(
        self,
        session: Session,
        now: datetime | None = None,
    ) -> bool:
        """
        Determine whether a session has exceeded its
        configured inactivity timeout.

        A timeout of zero disables inactivity expiration.

        Locked sessions are never considered newly expired.
        """

        if session.is_locked:
            return False

        timeout_seconds = (
            self._config.inactivity_timeout_seconds
        )

        if timeout_seconds == 0:
            return False

        current_time = (
            now
            if now is not None
            else self._now()
        )

        elapsed = (
            current_time
            - session.last_activity_at
        ).total_seconds()

        return elapsed >= timeout_seconds

    def ensure_valid(
        self,
        session: Session,
    ) -> Session:
        """
        Ensure the session has not expired.

        If inactivity expiration has occurred, the session
        is transitioned to LOCKED through expire().
        """

        if session.is_locked:
            return session

        if self.is_expired(session):
            self.expire(session)

        return session

    def expire(
        self,
        session: Session,
    ) -> Session:
        """
        Permanently expire a session due to inactivity.

        Expiration is terminal because it transitions the
        session to LOCKED.
        """

        if session.is_locked:
            return session

        reason = (
            "Session expired due to inactivity."
        )

        self.lock(
            session,
            reason=reason,
            audit_event_type=(
                AuditEventType.SESSION_EXPIRED
            ),
        )

        return session

    # ======================================================
    # MARK REVERIFY
    # ======================================================

    def mark_reverify(
        self,
        session: Session,
        *,
        reason: str | None = None,
    ) -> Session:
        if session.is_locked:
            return session

        now = self._now()

        previous_state = session.state

        session.state = (
            SessionState.REVERIFY
        )

        session.reverify_count += 1

        session.state_changed_at = now

        session.last_activity_at = now

        if previous_state != SessionState.REVERIFY:
            self._audit(
                AuditEventType.REVERIFICATION_REQUIRED,
                session=session,
                action="reverify",
                reason=reason,
            )

        return session

    # ======================================================
    # ACTIVATE
    # ======================================================

    def activate(
        self,
        session: Session,
        *,
        reason: str | None = None,
    ) -> Session:
        """
        Activate a session.

        Locked sessions are terminal and cannot be activated.
        """

        if session.is_locked:
            return session

        now = self._now()

        previous_state = session.state

        session.state = (
            SessionState.ACTIVE
        )

        session.lock_reason = None

        session.state_changed_at = now

        session.last_activity_at = now

        if previous_state != SessionState.ACTIVE:
            self._audit(
                AuditEventType.SESSION_ACTIVATED,
                session=session,
                action="activate",
                reason=(
                    reason
                    or
                    "Session activated."
                ),
            )

        return session

    # ======================================================
    # LOCK
    # ======================================================

    def lock(
        self,
        session: Session,
        reason: str,
        *,
        audit_event_type: AuditEventType = (
            AuditEventType.SESSION_LOCKED
        ),
    ) -> Session:
        """
        Permanently lock a session.

        LOCKED is terminal. A locked session cannot later be
        activated or restored.
        """

        if session.is_locked:
            return session

        now = self._now()

        session.state = (
            SessionState.LOCKED
        )

        session.lock_reason = reason

        session.state_changed_at = now

        session.last_activity_at = now

        self._audit(
            audit_event_type,
            session=session,
            action="lock",
            reason=reason,
        )

        return session

    # ======================================================
    # LOGOUT
    # ======================================================

    def logout(
        self,
        session: Session,
    ) -> Session:
        """
        Permanently terminate an authenticated session.

        Logout is intentionally terminal and uses the same
        server-side LOCKED state so an already issued session
        identifier can never be reused to regain access.

        The distinction between logout and security locking is
        preserved through the audit event and lock reason.
        """

        if session.is_locked:
            return session

        reason = (
            "Session logged out by the authenticated user."
        )

        return self.lock(
            session,
            reason=reason,
            audit_event_type=(
                AuditEventType.SESSION_LOGGED_OUT
            ),
        )

    # ======================================================
    # APPLY POLICY ACTION
    # ======================================================

    def apply_action(
        self,
        session: Session,
        action: DecisionAction | str,
        reason: str,
    ) -> Session:
        if isinstance(action, str):
            action = DecisionAction(
                action
            )

        if session.is_locked:
            return session

        if action == DecisionAction.ALLOW:
            self.activate(
                session,
                reason=reason,
            )

            return session

        if action == DecisionAction.REVERIFY:
            self.mark_reverify(
                session,
                reason=reason,
            )

            return session

        if action == DecisionAction.LOCK:
            self.lock(
                session,
                reason=reason,
            )

            return session

        raise ValueError(
            f"Unsupported session action: {action}"
        )

    # ======================================================
    # DELETE
    # ======================================================

    def delete(
        self,
        session_id: str,
    ) -> None:
        self._sessions.pop(
            session_id,
            None,
        )

    # ======================================================
    # CLEAR
    # ======================================================

    def clear(self) -> None:
        self._sessions.clear()

    # ======================================================
    # COUNT
    # ======================================================

    def count(self) -> int:
        return len(
            self._sessions
        )

    store, audit_logger, session = (
        create_store(
            timeout_seconds=60
        )
    )

    future = (
        session.last_activity_at
        + timedelta(seconds=61)
    )

    assert store.is_expired(
        session,

    )

    store.ensure_valid(
        session
    )

    # ensure_valid uses the real current time, so
    # explicitly expire for the deterministic lifecycle test.
    store.expire(
        session
    )

    assert session.state == (

    )

    assert session.is_locked

    assert session.lock_reason == (

    )
